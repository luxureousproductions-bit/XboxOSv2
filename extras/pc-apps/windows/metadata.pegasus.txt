# XboxOSv2 — PC Apps for the App Drawer (Windows)
#
# Put this folder in one of your Pegasus game folders (or add this folder in
# Pegasus › Settings › Game directories), then restart Pegasus.
# Lines starting with # are switched off: delete the # in front of an entry's
# lines to switch it on, and fix the path if yours is different.
#
# The genre picks the drawer tab:
#   Game · Emulator · Launcher · App · Tool  — and Power for the Power tile.
#
# Apps open through `cmd /c start`, so Pegasus comes straight back instead of
# waiting for the app to close, and %LOCALAPPDATA% style paths work.

collection: PC Apps
shortname: pcapps

# Every entry needs a `file:` so Pegasus keeps it. `pcapps:<name>` is a
# placeholder (no real file needed) — keep each one different.

# ── Tools (on every Windows PC) ────────────────────────────────────────────
game: File Explorer
file: pcapps:file-explorer
assets.boxFront: icons/file-explorer.png
genre: Tool
launch: cmd /c start "" explorer.exe

game: Task Manager
file: pcapps:task-manager
assets.boxFront: icons/task-manager.png
genre: Tool
launch: cmd /c start "" taskmgr.exe

game: Windows Settings
file: pcapps:windows-settings
assets.boxFront: icons/windows-settings.png
genre: Tool
launch: cmd /c start "" ms-settings:
