# XboxOSv2 — PC Apps for the App Drawer (Linux)
#
# Put this folder in one of your Pegasus game folders (or add this folder in
# Pegasus › Settings › Game directories), then restart Pegasus.
# Lines starting with # are switched off: delete the # in front of an entry's
# lines to switch it on, and change the command if yours is different.
#
# The genre picks the drawer tab:
#   Game · Emulator · Launcher · App · Tool  — and Power for the Power tile.
#
# Apps open through `setsid -f`, so Pegasus comes straight back instead of
# waiting for the app to close. Installed from Flathub? Use the
# `flatpak run …` lines instead.

collection: PC Apps
shortname: pcapps

# Every entry needs a `file:` so Pegasus keeps it. `pcapps:<name>` is a
# placeholder (no real file needed) — keep each one different.

# ── Tools ──────────────────────────────────────────────────────────────────
game: Files
file: pcapps:files
assets.boxFront: icons/files.png
genre: Tool
launch: setsid -f xdg-open .
